CREATE OR REPLACE FUNCTION get_road_status (
    p_pcr   NUMBER,
    p_iri   NUMBER,
    p_rut   NUMBER,
    p_crack NUMBER
) RETURN VARCHAR2 IS
    v_status VARCHAR2(20);
BEGIN
    IF
        p_pcr >= 80
        AND p_iri <= 3
        AND p_rut <= 3
        AND p_crack <= 10
    THEN
        v_status := 'GOOD';
    ELSIF p_pcr >= 60 THEN
        v_status := 'FAIR';
    ELSE
        v_status := 'POOR';
    END IF;

    RETURN v_status;
END;
/

SELECT
    get_road_status(90, 2.1, 1.5, 5) AS status
FROM
    dual;
