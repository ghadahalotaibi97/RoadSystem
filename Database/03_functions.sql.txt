CREATE OR REPLACE FUNCTION GET_ROAD_STATUS
(
P_PCR NUMBER,
P_IRI NUMBER,
P_RUT NUMBER,
P_CRACK NUMBER
)

RETURN VARCHAR2

IS

V_STATUS VARCHAR2(20);

BEGIN


IF P_PCR >= 80
AND P_IRI <= 3
AND P_RUT <= 3
AND P_CRACK <= 10
THEN

V_STATUS := 'GOOD';


ELSIF P_PCR >= 60
THEN

V_STATUS := 'FAIR';


ELSE

V_STATUS := 'POOR';


END IF;



RETURN V_STATUS;


END;
/
SELECT
GET_ROAD_STATUS
(
90,
2.1,
1.5,
5
)
AS STATUS

FROM DUAL;
