CREATE TABLE USERS
(
USER_ID NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
USERNAME VARCHAR2(50) UNIQUE NOT NULL,
PASSWORD VARCHAR2(100) NOT NULL,
ROLE VARCHAR2(30)
);

CREATE TABLE ROADS
(
ROAD_ID NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,

ROAD_NO VARCHAR2(20) NOT NULL,

ROAD_NAME VARCHAR2(100),

ROAD_TYPE VARCHAR2(30),

LENGTH_KM NUMBER(10,2),

CREATED_DATE DATE DEFAULT SYSDATE
);


CREATE TABLE SURVEY
(
SURVEY_ID NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,

ROAD_ID NUMBER,

SURVEY_DATE DATE,

PCR NUMBER(5,2),

IRI NUMBER(5,2),

RUT NUMBER(5,2),

CRACK NUMBER(5,2),


CONSTRAINT FK_SURVEY_ROAD
FOREIGN KEY(ROAD_ID)
REFERENCES ROADS(ROAD_ID)

);

CREATE TABLE ROAD_STATUS
(

STATUS_ID NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,


ROAD_ID NUMBER,


STATUS VARCHAR2(20),


CALC_DATE DATE DEFAULT SYSDATE,


CONSTRAINT FK_STATUS_ROAD
FOREIGN KEY(ROAD_ID)
REFERENCES ROADS(ROAD_ID)

);
