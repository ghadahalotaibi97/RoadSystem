CREATE TABLE users (
    user_id  NUMBER
        GENERATED BY DEFAULT AS IDENTITY
    PRIMARY KEY,
    username VARCHAR2(50) UNIQUE NOT NULL,
    password VARCHAR2(100) NOT NULL,
    role     VARCHAR2(30)
);

CREATE TABLE roads (
    road_id      NUMBER
        GENERATED BY DEFAULT AS IDENTITY
    PRIMARY KEY,
    road_no      VARCHAR2(20) NOT NULL,
    road_name    VARCHAR2(100),
    road_type    VARCHAR2(30),
    length_km    NUMBER(10, 2),
    created_date DATE DEFAULT sysdate
);

CREATE TABLE survey (
    survey_id   NUMBER
        GENERATED BY DEFAULT AS IDENTITY
    PRIMARY KEY,
    road_id     NUMBER,
    survey_date DATE,
    pcr         NUMBER(5, 2),
    iri         NUMBER(5, 2),
    rut         NUMBER(5, 2),
    crack       NUMBER(5, 2),
    CONSTRAINT fk_survey_road FOREIGN KEY ( road_id )
        REFERENCES roads ( road_id )
);

CREATE TABLE road_status (
    status_id NUMBER
        GENERATED BY DEFAULT AS IDENTITY
    PRIMARY KEY,
    road_id   NUMBER,
    status    VARCHAR2(20),
    calc_date DATE DEFAULT sysdate,
    CONSTRAINT fk_status_road FOREIGN KEY ( road_id )
        REFERENCES roads ( road_id )
);
