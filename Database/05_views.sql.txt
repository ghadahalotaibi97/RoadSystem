CREATE OR REPLACE VIEW road_report AS
    SELECT
        r.road_no,
        r.road_name,
        r.road_type,
        s.pcr,
        s.iri,
        s.rut,
        get_road_status(s.pcr, s.iri, s.rut, s.crack) status
    FROM
             roads r
        JOIN survey s ON r.road_id = s.road_id;
