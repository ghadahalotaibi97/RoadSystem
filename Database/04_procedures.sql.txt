CREATE OR REPLACE PROCEDURE ADD_SURVEY
(
P_ROAD_ID NUMBER,
P_PCR NUMBER,
P_IRI NUMBER,
P_RUT NUMBER,
P_CRACK NUMBER
)

IS

V_STATUS VARCHAR2(20);


BEGIN


INSERT INTO SURVEY
(
ROAD_ID,
SURVEY_DATE,
PCR,
IRI,
RUT,
CRACK
)

VALUES
(
P_ROAD_ID,
SYSDATE,
P_PCR,
P_IRI,
P_RUT,
P_CRACK
);



V_STATUS :=
GET_ROAD_STATUS
(
P_PCR,
P_IRI,
P_RUT,
P_CRACK
);



INSERT INTO ROAD_STATUS
(
ROAD_ID,
STATUS
)

VALUES
(
P_ROAD_ID,
V_STATUS
);


COMMIT;


END;
/
BEGIN

);

END;
