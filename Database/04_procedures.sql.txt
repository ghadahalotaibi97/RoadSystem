CREATE OR REPLACE PROCEDURE add_survey (
    p_road_id NUMBER,
    p_pcr     NUMBER,
    p_iri     NUMBER,
    p_rut     NUMBER,
    p_crack   NUMBER
) IS
    v_status VARCHAR2(20);
BEGIN
    INSERT INTO survey (
        road_id,
        survey_date,
        pcr,
        iri,
        rut,
        crack
    ) VALUES ( p_road_id,
               sysdate,
               p_pcr,
               p_iri,
               p_rut,
               p_crack );

    v_status := get_road_status(p_pcr, p_iri, p_rut, p_crack);
    INSERT INTO road_status (
        road_id,
        status
    ) VALUES ( p_road_id,
               v_status );

    COMMIT;
END;
/

BEGIN
    add_survey(1, 85, 2, 2, 6);
END;
